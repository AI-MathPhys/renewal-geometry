\section{Operational data}
The programme begins with a repeated finite experiment rather than with a manifold, gauge group, or spectral triple.  The minimal abstract data needed for the matter chain are collected here.

\begin{definition}[Renewal experiment]
\label{def:renewal-experiment}
A \emph{renewal experiment} is a tuple
\[
 \mathfrak R=(\mathfrak A,\rho_*,\{\mathcal I_\omega\}_{\omega\in\Omega},\Theta,\mathscr R)
\]
where:
\begin{enumerate}[label=\textup{(R\arabic*)},leftmargin=2.8em]
\item $\mathfrak A$ is a finite or semifinite observable algebra;
\item $\rho_*$ is a faithful stationary reference state;
\item $\{\mathcal I_\omega\}$ is an instrument of completely positive continuation maps, with $\sum_\omega\mathcal I_\omega$ trace preserving;
\item $\Theta$ is a renewal shift which restarts the experiment after a completed private event;
\item $\mathscr R$ is a microscopic reversal or modular conjugation compatible with $\rho_*$.
\end{enumerate}
The instrument is \emph{KMS symmetric} when its continuous-time envelope has a generator self-adjoint in the KMS inner product associated with $\rho_*$.  It is \emph{predictively finite} when the future-equivalence quotient of past histories generates a finite sufficient algebra.
\end{definition}

This definition is intentionally broader than the explicit six-state model used later.  The upstream matter theorem needs an operator-valued renewal experiment with a first-order calculus; the explicit source presently constructed supplies only part of that package.

\begin{definition}[Predictive equivalence and sufficient algebra]
Two past histories $h,h'$ are predictively equivalent when every future continuation word has the same conditional probability and the same retained quantum output from $h$ and $h'$.  The quotient classes generate the finite predictive algebra $\mathfrak A_F$.  If $E_F:\mathfrak A\to\mathfrak A_F$ is the $\rho_*$-preserving conditional expectation, then $\mathfrak A_F$ is called \emph{sufficient} when relative entropy is preserved for the selected family of states.
\end{definition}

The use of sufficient subalgebras and conditional expectations is standard in quantum statistical inference; see Petz's sufficiency theorems and the Accardi--Cecchini construction \cite{Petz1986,Petz1988,AccardiCecchini1982,Takesaki2003}.  The renewal interpretation adds an operational meaning: $\mathfrak A_F$ is the smallest algebra whose record suffices for every future test in the retained experiment.

\section{Finite selector theorem}

\begin{theorem}[Finite predictive selector]
\label{thm:finite-selector}
Let $\{\rho_\theta\}_{\theta\in\Theta}$ be a finite-dimensional faithful statistical model with barycentre
\[
 \bar\rho=\sum_{\theta}p_\theta\rho_\theta>0.
\]
Let
\[
 h_\theta=\bar\rho^{-1/2}\rho_\theta\bar\rho^{-1/2}
\]
be the normalized likelihood operators, and let $\mathfrak N$ be the unital $*$-algebra they generate.  If $\mathfrak N$ is invariant under the modular group of $\bar\rho$, then there exists a unique $\bar\rho$-preserving conditional expectation
\[
 E_{\bar\rho}:M_n(\C)\longrightarrow\mathfrak N.
\]
Moreover, $\mathfrak N$ is sufficient for the model and is minimal among modularly invariant sufficient subalgebras containing the likelihood operators.
\end{theorem}

\begin{proof}
Modular invariance permits the Takesaki conditional expectation onto $\mathfrak N$ \cite{Takesaki2003,AccardiCecchini1982}.  Since every $h_\theta$ belongs to $\mathfrak N$,
\[
 \rho_\theta=\bar\rho^{1/2}h_\theta\bar\rho^{1/2}
\]
is reconstructed from the restriction of the state to $\mathfrak N$.  Petz's equality criterion for monotonicity of relative entropy then gives sufficiency.  If $\mathfrak M$ is another modularly invariant sufficient algebra, the Petz recovery identity forces each normalized likelihood operator $h_\theta$ to belong to $\mathfrak M$.  Hence $\mathfrak N\subseteq\mathfrak M$.
\end{proof}

\Exact{The theorem is an ordinary finite-dimensional result.  Its renewal use is conditional on having exported the correct likelihood family and on verifying modular invariance for the actual microscopic source.}

\section{Likelihood cocycle and the renewal clock}
For a scalar acceptance/rejection opportunity with acceptance probability $\vartheta$, an exponential tilt by $q$ has partition function
\[
 Z(q)=1-\vartheta+\vartheta e^q.
\]
If $N_n$ counts accepted responses in $n$ completed opportunities, then
\[
 \frac{\dd\mathbb P_q}{\dd\mathbb P_0}
 =\exp\bigl(qN_n-n\log Z(q)\bigr),
 \qquad
 \mathfrak M_n(q)=\frac{e^{qN_n}}{Z(q)^n}
\]
is a positive mean-one martingale.  This elementary cocycle was made exact in the archive for the accepted-response instrument.

\begin{proposition}[Renewal subordination of the counting pressure]
\label{prop:renewal-pressure}
Let $W$ be the completed-private waiting time with probability generating function $F_W(z)$.  If each completed opportunity is accepted with tilted weight $Z(q)$, the generating function of the accepted-count moment is
\[
 \sum_{t\ge0}\mathbb E\,e^{qN_{\rm acc}(t)}z^t
 =\frac{1-F_W(z)}{(1-z)(1-Z(q)F_W(z))}.
\]
The exponential pressure per raw renewal tick is determined by the smallest positive singularity $\rho(q)$ solving
\[
 Z(q)F_W(\rho(q))=1,
 \qquad
 \psi_{\rm raw}(q)=-\log\rho(q).
\]
\end{proposition}

\begin{proof}
Decompose the path at the last completed renewal.  The factor $(1-F_W(z))/(1-z)$ is the generating function of the terminal incomplete waiting interval, and every completed interval contributes $F_W(z)$ together with the acceptance moment $Z(q)$.  Summing the geometric series gives the displayed identity.  Standard singularity analysis of positive generating functions gives the pressure.
\end{proof}

\begin{warning}
The scalar counting cocycle is not a matter source.  In the explicit accepted-response instrument all accepted labels have the same state-independent likelihood score.  Its horizontal Choi derivative has no finite-charge support rotation.  It calibrates a dimensionless renewal clock, not a Yukawa or Higgs amplitude.
\end{warning}
