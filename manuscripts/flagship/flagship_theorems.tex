\begin{definition}[Reflected zero heat]

The reflection symmetry of the zero set makes $u_\chi$ real valued.

\begin{theorem}[Gaussian heat criterion for Dirichlet GRH]
\label{thm:GRH-heat-master}
For a primitive Dirichlet character $\chi$, the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.7em]
\item every nontrivial zero of $\Xi_\chi$ has real part $1/2$;
\item $u_\chi(t,\tau)\ge0$ for every $t>0$ and $\tau\in\R$;
\item the reflected $2\times2$ affine moment matrix generated by the ports
      $1$ and $s-1/2$ is positive semidefinite at every Gaussian scale and
      center;
\item
      \[
       u_\chi\,\partial_\tau^2u_\chi
       -(\partial_\tau u_\chi)^2
       +\frac1{2t}u_\chi^2\ge0
      \]
      together with $u_\chi\ge0$ for every $t,\tau$.
\end{enumerate}
For the principal zeta channel, the pole contribution must be cleared exactly
before applying the criterion.  RH is the corresponding principal-channel
specialization; GRH is the conjunction over the chosen primitive character
family.
\end{theorem}

\begin{proof}[Proof from Gaussian orbit exposure]
If all zeros lie on the line, every summand is the ordinary positive heat
kernel centered at $\Im\rho$, proving positivity.  Conversely, an off-line
reflection pair $1/2\pm c+i\gamma$ contributes
\[
 \frac{2}{\sqrt{4\pi t}}
 e^{(c^2-x^2)/(4t)}\cos\!\left(\frac{cx}{2t}\right),
 \qquad x=\tau-\gamma.
\]
Choosing $x=2\pi t/c$ makes this contribution negative with exponential
magnitude $e^{c^2/(4t)+O(t)}$.  A Gaussian outer-exposure argument chooses an
off-line orbit of maximal horizontal displacement and a sequence of centers
for which all competing orbits are exponentially smaller.  Positivity then
fails.  This proves \textup{(i)}$\Leftrightarrow$\textup{(ii)}.
The affine matrix is the Gram of $1$ and $s-1/2$ under the reflected zero
functional.  Direct differentiation of the Gaussian gives its determinant,
up to a positive scalar, as the displayed heat-semiconvexity expression,
proving the remaining equivalences.
\end{proof}

where $a_m$ and $b_n$ are the displayed heat-Mellin von Mangoldt

\end{theorem}

\begin{proof}

\begin{theorem}[Twin-prime heat sufficiency]

The divergence certificate is not proved by the universal Grand Tensor.

\begin{proof}

 O\!\left(N^{-1/4}(\log N)^2\right).

\begin{proof}

\subsection{Yang--Mills: physical loop-complexity regeneration}

Let $T=e^{-\tau H}$ be a positive centered gauge-invariant transfer

 \qquad a=\norm A,\quad b=\norm B.

\[
 \mathcal H_{\rm neu}=\bigoplus_{j\ge0}\mathcal E_j,
 \qquad \Pi_j:\mathcal H_{\rm neu}\to\mathcal E_j,

whose labels are retained loop-size, representation, multi-loop, center, or
cohomology records.  It is nearest-neighbour when
\[

For a finite source $S:\C^m\to\mathcal E_0$ put
\[
 \mathcal K_n(S)=\Span\{T^k\Ran S:0\le k<n\},
 \qquad
 \mathcal G_n=\bigoplus_{j=0}^{n-1}\mathcal E_j,
\]
and let $P_n$ be the projection onto $\mathcal K_n(S)$.

\begin{theorem}[Physical-shell localization of the Hankel source]
\label{thm:YM-physical-shell-master}
If the physical filtration is nearest-neighbour, then for every $n\ge1$,
\[
 \boxed{\mathcal K_n(S)\subseteq\mathcal G_n,}
\]
and for the exact Hankel leakage $R_n=(I-P_n)TP_n$,
\[
 \boxed{\Ran R_n\subseteq
 \mathcal G_{n+1}\ominus\mathcal K_n(S).}
\]
If the predeclared physical shell atlas is source cyclic through level $n$,
so that $\mathcal K_n(S)=\mathcal G_n$, then
\[
 \boxed{\Ran R_n\subseteq\mathcal E_n.}
\]
If, in addition, every resolved one-slab history enlarges geometric support by
at most a fixed collar $R_0$, the first residual is supported in the
$R_0$-collar of the initial source and the $n$th physical shell in its
$nR_0$-collar.  Thus the first missing source is a predeclared local physical
history, not merely an abstract Ritz vector.
\end{theorem}

\begin{proof}
Since $\Ran S\subseteq\mathcal E_0$, the first inclusion holds for $n=1$.
If $T^k\Ran S\subseteq\mathcal G_{k+1}$, nearest-neighbour propagation gives
$T^{k+1}\Ran S\subseteq T\mathcal G_{k+1}\subseteq\mathcal G_{k+2}$.
Induction proves $\mathcal K_n(S)\subseteq\mathcal G_n$.  Hence
$TP_n\mathcal H\subseteq\mathcal G_{n+1}$; subtracting its component in
$\mathcal K_n(S)$ proves the leakage inclusion.  Under
$\mathcal K_n(S)=\mathcal G_n$, the orthogonal difference
$\mathcal G_{n+1}\ominus\mathcal G_n$ is $\mathcal E_n$.  The collar statement
follows by the same induction applied to geometric supports.
\end{proof}

Let a resolved complexity process have growth, reduction, stay, and direct
regeneration probabilities
\[
 p,q,r,\kappa\ge0,
 \qquad p+q+r+\kappa=1.
\]
At a finite tail cutoff $N$, let $K_N^{p,q,r}$ be the symmetric tridiagonal
matrix with
\[
 (K_N)_{jj}=r\ (j<N),
 \qquad (K_N)_{NN}=r+p,
 \qquad (K_N)_{j,j+1}=\sqrt{pq}.
\]
A positive operator-valued tail $D_N$ is a KMS fibre extension of this process
when
\[
 \norm{\Pi_iD_N\Pi_j}\le (K_N^{p,q,r})_{ij}.
\]

\begin{theorem}[Uniform KMS birth--death tail bound]
\label{thm:YM-birth-death-tail-master}
Every such KMS fibre extension satisfies
\[
 \boxed{\norm{D_N}\le1-\kappa.}
\]
If $q\ge p$, then the stronger uniform estimate holds:
\[
 \boxed{
 \norm{D_N}\le r+2\sqrt{pq}
 =1-\kappa-(\sqrt q-\sqrt p)^2.}
\]
Both estimates are independent of the maximal complexity $N$ and of all
internal shell multiplicities.
\end{theorem}

\begin{proof}
For $x=\sum_jx_j$ with $x_j\in\mathcal E_j$, put $y_j=\norm{x_j}$.  Positivity
and the block bounds give
\[
 \langle x,D_Nx\rangle
 \le\sum_{i,j}(K_N)_{ij}y_iy_j
 \le\norm{K_N}\norm{x}^2,
\]
so $\norm{D_N}\le\norm{K_N}$.  For $p,q>0$, let
$t=\sqrt{p/q}$ and $w_j=t^j$.  Every interior and reflected top row satisfies
$(K_Nw)_j/w_j=1-\kappa$, while the bottom quotient is at most this value.
Collatz--Wielandt therefore gives $\rho(K_N)\le1-\kappa$.  Since $K_N$ is real
symmetric and entrywise nonnegative, $\norm{K_N}=\rho(K_N)$.  The degenerate
cases follow by continuity.  If $q\ge p$, every row sum is at most
$r+2\sqrt{pq}$, including the reflected top row because
$p+\sqrt{pq}\le2\sqrt{pq}$.  The row-sum bound gives the second estimate, and
$1-r-2\sqrt{pq}=\kappa+(\sqrt q-\sqrt p)^2$.
\end{proof}

\begin{countertheorem}[Conservative loop diffusion cannot produce a uniform mass]
\label{cth:YM-loop-diffusion-master}
Assume $\kappa=0$.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.7em]
\item If $p=q>0$, then
      \[
       \lim_{N\to\infty}\norm{K_N^{p,p,1-2p}}=1.
      \]
\item If $p>q\ge0$, then
      \[
       \lim_{N\to\infty}\norm{K_N^{p,q,1-p-q}}=1.
      \]
\end{enumerate}
Consequently, in the absence of direct regeneration, a volume-uniform
birth--death tail gap requires strict return bias $q>p$.
\end{countertheorem}

\begin{proof}
For $p=q$, take a unit vector approximately constant on a growing interval of
shells away from the lower boundary.  All interior rows have sum one and only
the two ends of the interval contribute a defect, so its Rayleigh quotient
tends to one.  For $p>q>0$, take $w_j=(p/q)^{j/2}$.  One has $K_Nw=w$ on every
row except the bottom row, while $\norm w$ is dominated by the top shells;
therefore $\norm{(K_N-I)w}/\norm w\to0$.  If $q=0$, a vector supported at the
reflected top shell gives the same conclusion.  Since every $K_N$ is a
contraction, the norms tend to one.
\end{proof}

Define the physical regeneration margin
\[
 \mu_{\rm reg}=
 \begin{cases}
 \kappa+(\sqrt q-\sqrt p)^2,&q\ge p,\\
 \kappa,&q<p.
 \end{cases}
\]
Let $D^{\rm bd}$ be the KMS fibre extension generated by the predeclared
complexity record and let
\[
 \delta_{\rm comp}=\norm{D-D^{\rm bd}},
 \qquad
 \mu_{\rm eff}=\mu_{\rm reg}-\delta_{\rm comp}.
\]
Then \Cref{thm:YM-birth-death-tail-master} and the triangle inequality give
\[
 \norm D\le1-\mu_{\rm eff}.
\]
The defect $\delta_{\rm comp}$ measures unresolved physical histories rather
than a fit to the low eigenvalue; a large defect requires enlargement of the
source atlas.

\begin{theorem}[Explicit regulated neutral mass criterion]
\label{thm:YM-regulated-master}
If
\[
 a<1,\qquad \mu_{\rm eff}>0,
 \qquad b^2<(1-a)\mu_{\rm eff},
\]
then
\[
 \boxed{
 \norm T\le q_{\rm reg}
 :=\frac{a+1-\mu_{\rm eff}
 +\sqrt{(a-1+\mu_{\rm eff})^2+4b^2}}2<1,}
\]
and
\[
 \boxed{\Delta_{\rm reg}\ge-\tau^{-1}\log q_{\rm reg}>0.}
\]
If the invariant-loop source family is future cyclic and the constants are
uniform in volume and lattice spacing, the bound holds on the complete
centered finite-lattice gauge-invariant Hilbert space.
\end{theorem}

\begin{proof}
The uniform tail theorem and the source-defect estimate above give
$\norm D\le1-\mu_{\rm eff}$.  The norm of a positive $2\times2$ block
operator is bounded by the larger eigenvalue of the scalar matrix
$\begin{psmallmatrix}a&b\\b&1-\mu_{\rm eff}\end{psmallmatrix}$, which is
$q_{\rm reg}$.  The strict determinant condition is exactly
$b^2<(1-a)\mu_{\rm eff}$ and places the larger root below one.  The transfer
relation $T=e^{-\tau H}$ gives the mass bound.
\end{proof}

\subsection{Birch--Swinnerton--Dyer: analytic jets and the missing comparison}

\begin{theorem}[Exact analytic-rank selector]
\label{thm:BSD-jet-master}
If $r=\operatorname{ord}_{z=0}F$, then
\[
 \boxed{\dim\Ker\mathbb J_m(F)=\min\{r,m\}.}

For an elliptic curve $E/\mathbb Q$ and

\begin{proof}
Write $F=z^rg$ with $g(0)\ne0$.  Multiplication by $g$ is invertible in the
local algebra, so the kernel is that of multiplication by $z^r$, namely the
span of $z^{m-r},\ldots,z^{m-1}$ when $r<m$ and the whole space when

\end{proof}

\label{thm:BSD-bridge-master}

\begin{proof}

two minimal realizations and therefore equality of their distinguished zero
multiplicities.  The jet nullity is $r_{\rm an}$ and the independent

\subsection{Hodge: harmonic shorting and cycle observability}

\begin{theorem}[Hodge cycle-observability equivalence]
\label{thm:Hodge-master}

\[
 \mathcal Z_X:\C^m\to H_{\rm Hdg}^{2p}(X;\C),

\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.7em]
\item the classes $[Z_j]$ span the Hodge sector;
\item $\mathcal Z_X$ is surjective;
\item $\mathcal Z_X\mathcal Z_X^*$ is positive definite on the Hodge sector;
\item $\mathcal Z_X\mathcal Z_X^*\succeq c_XI$ for some $c_X>0$.
\end{enumerate}

family of rational cycles satisfying these conditions.  Shorting exact and
coexact forms gives harmonic projection but does not by itself prove cycle
surjectivity.
\end{theorem}

\begin{proof}
In finite dimensions, spanning is surjectivity, which is equivalent to
injectivity of $\mathcal Z_X^*$.  Since
$\langle h,\mathcal Z_X\mathcal Z_X^*h\rangle=
\norm{\mathcal Z_X^*h}^2$, this is positive definiteness, equivalent to a
positive least eigenvalue.  A finite rational spanning family is exactly the
fixed-sector Hodge assertion.
\end{proof}

 \operatorname{CircuitSize}(f)\ge R_n(f)/p(n)

\begin{lemma}[Common-core strong-resolvent criterion]
\label{lem:app-core-resolvent}
Let $A_n$ and $A$ be self-adjoint operators on one Hilbert space $\Hh$.  Let
$\mathcal C\subset\Dom A$ be a core for $A$, assume
$\mathcal C\subset\Dom A_n$ for every $n$, and suppose
\[
  A_nu\longrightarrow Au
  \qquad (u\in\mathcal C).
\]
Then $A_n\to A$ in strong-resolvent sense.  Consequently
$f(A_n)\to f(A)$ strongly for every $f\in C_0(\R)$, and
$e^{-itA_n}\to e^{-itA}$ strongly, uniformly for $t$ in compact intervals.
\end{lemma}

\begin{proof}
Fix $z\in\C\setminus\R$.  The map
$A-z:\Dom A\to\Hh$ is continuous for the graph norm and is onto because $A$
is self-adjoint.  Since $\mathcal C$ is graph-norm dense in $\Dom A$, its image
$(A-z)\mathcal C$ is therefore dense in $(A-z)\Dom A=\Hh$.

For $y=(A-z)u$ with $u\in\mathcal C$, the resolvent identity on the vector $u$
gives
\[
 (A_n-z)^{-1}y-u
 =(A_n-z)^{-1}(A-A_n)u.
\]
Since $\|(A_n-z)^{-1}\|\le |\im z|^{-1}$ and $(A_n-A)u\to0$, the right-hand
side tends to zero.  But $u=(A-z)^{-1}y$, so the resolvents converge on the
dense set $(A-z)\mathcal C$.  The same uniform resolvent bound extends the
convergence to every $y\in\Hh$.  This is strong-resolvent convergence.  The
$C_0$ functional calculus and compact-time unitary-group convergence are the
standard spectral-theorem consequences of strong-resolvent convergence.
\end{proof}

\label{app:proof-predictive-geometry}

\subsection{Universal future quotient}

For a rooted history $h:I\to B$, define its future profile by
\[
  \Pi_B(h)(r,e):=\Pr(e\circ r\circ h).
\]
Future equivalence is equality of these functions.

\begin{lemma}[Future congruence and universal minimality]
\label{lem:app-future-minimality}
Future equivalence is an equivalence relation stable under admissible
postcomposition.  Its quotient $\Qq_+(B)$ is the unique reduced
future-sufficient memory, up to unique isomorphism fixing the encoded histories.
\end{lemma}

\begin{proof}
Equality of functions is an equivalence relation.  If $h\sim_+h'$ and
$k:B\to C$ is admissible, then for every later continuation $d$ and terminal
test $e$,
\[
 \Pr(e\circ d\circ k\circ h)
 =\Pr(e\circ(d\circ k)\circ h)
 =\Pr(e\circ(d\circ k)\circ h')
 =\Pr(e\circ d\circ k\circ h'),
\]
so $k\circ h\sim_+k\circ h'$.

Let $X(B)$ be any future-sufficient realization with surjective encoding
$\eta_B$.  Define
\[
  \pi_B^X(\eta_B(h)):=[h].
\]
If $\eta_B(h)=\eta_B(h')$, transition compatibility and equality of all output
functions imply $\Pi_B(h)=\Pi_B(h')$, so the formula is well defined.  It is
surjective and intertwines all continuations and outputs.  If $X$ is reduced,
equal quotient classes have equal future outputs and hence represent the same
point of $X$, so $\pi_B^X$ is injective.  Conversely injectivity of $\pi_B^X$
implies reducedness.  Uniqueness follows from surjectivity of the encoding.
\end{proof}

\subsection{Conditional successors and scalar renewal}

For a successful resolved history $h$, put
$p(h)=\Pr(u\circ h)>0$ and
\[
  \Pi^{\rm cond}(h)(F):=\frac{\Pr(F\circ h)}{p(h)}.
\]

\begin{lemma}[Resolved successor congruence]
\label{lem:app-successor}
If $h$ and $h'$ have the same conditional future profile and a resolved branch
$e$ succeeds after both, then $e\circ h$ and $e\circ h'$ have the same
conditional future profile.
\end{lemma}

\begin{proof}
For every future test $F$ after $e$, conditional equivalence applied to
$F\circ e$ and to the discard after $e$ gives
\[
 \frac{\Pr(F\circ e\circ h)}{p(h)}
 =\frac{\Pr(F\circ e\circ h')}{p(h')},
 \qquad
 \frac{p(e\circ h)}{p(h)}
 =\frac{p(e\circ h')}{p(h')}.
\]
The second common value is positive.  Dividing the first equality by the second
is exactly conditional equivalence of the extended histories.
\end{proof}

It follows that the observed branch $e$ defines a partial successor map
$\delta_e([h])=[e\circ h]$ on the conditional quotient.

\begin{lemma}[Reference-renewal identity]
\label{lem:app-reference-renewal}
In a reduced future-tomographic operator realization, every positive-weight
resolved edge $e:x\to y=\delta_e(x)$ satisfies
\[
  \mathcal E_e(\rho_x)=w_e\rho_y,
  \qquad w_e=\Tr\mathcal E_e(\rho_x).
\]
Consequently the weight of a path is the product of its edge weights.
\end{lemma}

\begin{proof}
The normalized branch output
$\widehat\rho_e=\mathcal E_e(\rho_x)/w_e$ represents the extended resolved
history.  By \Cref{lem:app-successor}, that history lies in the unique successor
class $y$.  Thus $\widehat\rho_e$ and $\rho_y$ agree on every admitted future
effect.  Future tomography separates reachable normalized states, so they are
equal.  Multiplying by $w_e$ proves the one-edge identity.  Induction along
$e_n\cdots e_1$ gives
\[
 \mathcal E_{e_n}\cdots\mathcal E_{e_1}(\rho_{x_0})
 =\left(\prod_{j=1}^n w_{e_j}\right)\rho_{x_n}.
\]
\end{proof}

\subsection{Predictive units}

\begin{lemma}[A UCP unit is an automorphism]
\label{lem:app-ucp-unit}
If a UCP map $\Phi$ on a finite-dimensional $C^*$-algebra has a UCP inverse
$\Psi$, then $\Phi$ is a $*$-automorphism.
\end{lemma}

\begin{proof}
Kadison--Schwarz gives
$\Phi(a)^*\Phi(a)\le\Phi(a^*a)$.  Applying $\Psi$ gives
\[
 \Psi(\Phi(a)^*\Phi(a))\le a^*a.
\]
Kadison--Schwarz for $\Psi$ at $\Phi(a)$ gives the reverse inequality,
\[
 a^*a=\Psi(\Phi(a))^*\Psi(\Phi(a))
 \le\Psi(\Phi(a)^*\Phi(a)).
\]
Hence equality holds.  Applying the same argument to $aa^*$ places every $a$ in
the multiplicative domain of $\Phi$, so $\Phi$ is a unital $*$-homomorphism.
Bijectivity makes it a $*$-automorphism.
\end{proof}

Accumulated UCP maps are closed under composition and contain the identity, so
they form a monoid; \Cref{lem:app-ucp-unit} proves the unit claim.

\subsection{Predictive-length spectral triple}

Let $E$ be the finite reset alphabet and let $\W=E^*/\!\sim$ be the quotient by
equality of accumulated predictive channels.  Put
\[
 \Lambda([w])=1+\min\{|v|:v\sim w\},
 \qquad
 D_+e_{[w]}=\Lambda([w])e_{[w]}.
\]
For a letter $\sigma$, let
$S_\sigma e_{[w]}=e_{[\sigma w]}$.

\begin{lemma}[Bounded shifts, commutators, and compact resolvent]
\label{lem:app-positive-triple}
If every $S_\sigma$ has uniformly finite fibres and
$|\Lambda([\sigma w])-\Lambda([w])|$ is uniformly bounded, then the algebra
generated by the shifts, their adjoints, and finite-support diagonal
multipliers, together with $D_+$, is a spectral triple.  Its zeta abscissa is
\[
 q_{\pred}=\limsup_{R\to\infty}
 \frac{\log N(R)}{\log R},
 \qquad N(R)=\#\{[w]:\Lambda([w])\le R\}.
\]
\end{lemma}

\begin{proof}
Let $M_\sigma$ be the supremum of the cardinalities of the fibres of the
map $[w]\mapsto[\sigma w]$.  For every finitely supported vector $\xi$,
Cauchy--Schwarz on each fibre gives
\[
 \|S_\sigma\xi\|^2
 =\sum_u\left|\sum_{[w]:[\sigma w]=u}\xi_{[w]}\right|^2
 \le M_\sigma\sum_{[w]}|\xi_{[w]}|^2.
\]
Hence $S_\sigma$ extends boundedly with
$\|S_\sigma\|\le\sqrt{M_\sigma}$.  On basis vectors,
\[
 [D_+,S_\sigma]e_{[w]}
 =\bigl(\Lambda([\sigma w])-\Lambda([w])\bigr)e_{[\sigma w]},
\]
which is bounded by hypothesis.  The adjoints obey the same estimate.
For each $R$, every class in the length ball has a representative word of
length at most $R-1$, and there are finitely many such words; hence all spectral
projections of $D_+$ on bounded intervals have finite rank and the resolvent is
compact.

For the abscissa, let $q$ denote the displayed limsup.  If $s>q$, choose
$\epsilon>0$ with $s>q+2\epsilon$.  For all large $R$,
$N(R)\le R^{q+\epsilon}$.  Dyadic decomposition gives
\[
 \sum_{[w]}\Lambda([w])^{-s}
 \le C+\sum_{k\ge k_0}N(2^{k+1})2^{-ks}
 \le C+2^{q+\epsilon}\sum_{k\ge k_0}
 2^{-k(s-q-\epsilon)}<\infty.
\]
If $s<q$, choose $\epsilon>0$ with $s+\epsilon<q$ and a sequence $R_j\to\infty$
with $N(R_j)\ge R_j^{s+\epsilon}$.  The partial zeta sum over the length ball is
at least $N(R_j)R_j^{-s}\ge R_j^\epsilon\to\infty$.  Thus the abscissa is $q$.
\end{proof}

\subsection{Order and positivity boundary}

Define $a\preceq b$ in the predictive monoid if $b=ca$ for some accumulated
channel $c$.  This is a preorder.  Quotienting by
$a\approx b\Longleftrightarrow(a\preceq b\ \&\ b\preceq a)$ gives a partial
order.  On a component where $\Lambda$ is constant on $\approx$-classes and
strictly increases under strict divisibility, it descends to an intrinsic time
function.  This proves item \textup{(v)} of \Cref{thm:predictive-geometry}.

For item \textup{(vi)}, let $\mathfrak B_+$ be the irreducibly represented
positive renewal algebra.  A functorially extracted fundamental symmetry must
commute with $\mathfrak B_+$.  Since $\mathfrak B_+'=\C\one$, it is scalar;
self-adjointness and $J^2=\one$ give $J=\pm\one$, hence a definite form.
More generally, suppose a diagonal involution satisfies
$JS_e=c(e)S_eJ$ with $c(e)\in\{\pm1\}$.  Writing $Je_x=\eta(x)e_x$ gives
\[
  c(e:x\to y)=\eta(y)\eta(x).
\]
Thus the compatibility signs are an edgewise $\Ztwo$ cochain.  Changing local
sign conventions changes it by a vertex coboundary.  A nonconstant compatible
involution therefore contains precisely the extra signed information later
realized by the double cover.  Combining the preceding lemmas proves every item
of \Cref{thm:predictive-geometry}.

\subsection{Determinant orientation}

At each predictive vertex let $R_x^0$ be the real contrast space of the retained
future-distinct record algebra.  Along an oriented edge $e:x\to y$, determinant
regularity gives an isomorphism
$Q_e:R_x^0\to R_y^0$.  Choosing local orientations, define
\[
  \chi(e):=\sgn\det Q_e\in\{\pm1\}.
\]
Determinants multiply under path composition.  Reversing the chosen local
orientation at a vertex multiplies all incident signs by the corresponding
vertex coboundary, so the gauge-invariant class is
$[\chi]\in H^1(G;\Ztwo)$.

By \Cref{ass:finite-phase}\textup{(P6)}, some recurrent physical loop acts on
the two retained order records by their transposition.  On the one-dimensional
contrast line that action is multiplication by $-1$, so the determinant product
around the loop is $-1$.  Hence $[\chi]\ne0$.

\subsection{Minimal principal cover}

\begin{lemma}[Cone-positive minimal enrichment]
\label{lem:app-minimal-cover}
Over a fixed connected graph $G$, the gauge classes of objectwise-minimal,
cone-preserving, fibrewise-indefinite local enrichments are in bijection with
$H^1(G;\Ztwo)$.  Every such enrichment has fibre rank two.
\end{lemma}

\begin{proof}
Fibrewise indefiniteness requires at least one positive and one negative local
line, so the rank is at least two.  Rank two is realized by a signed double
cover.  A unitary carrying one distinguished simplicial renewal cone onto
another permutes its extremal rays: it maps each extremal ray to an extremal
ray, and unitarity makes the positive scalar on that ray equal to one.  Thus in
rank two every edge transport is either the identity or the sheet swap
$S=\begin{psmallmatrix}0&1\\1&0\end{psmallmatrix}$.

Choose sheet labels at the vertices and write the edge transport as
$S^{c(e)}$, $c(e)\in\F_2$.  A change of sheet label $a:V\to\F_2$ sends
$c\mapsto c+\delta a$.  Since a graph has no two-cells, every one-cochain is a
cocycle.  Therefore the gauge classes are
$C^1(G;\F_2)/\delta C^0(G;\F_2)=H^1(G;\Ztwo)$.  The representative $\chi$
constructs the cover
\[
  G^\chi:
  (x,\eta)\xrightarrow{e}(y,\eta\chi(e)),
  \qquad \eta\in\{\pm1\}.
\]
The sheet-sign operator $J_\chi e_{(x,\eta)}=\eta e_{(x,\eta)}$ is a
self-adjoint involution with one positive and one negative line in every fibre.
\end{proof}

Because $[\chi]\ne0$ and $G$ is connected, the monodromy map
$\pi_1(G)\to\Ztwo$ is surjective, so $G^\chi$ is connected.

\subsection{Amplitude lifts}

A connected finite graph is homotopy equivalent to a wedge of
$b_1(G)$ circles.  Hence
\[
 H^1(G;\Zfour)\cong(\Zfour)^{b_1(G)},
 \qquad
 H^1(G;\Ztwo)\cong(\Ztwo)^{b_1(G)}.
\]
Reduction modulo two is coordinatewise surjective, so every signed class has an
amplitude lift and exactly $2^{b_1(G)}$ such lifts.  Applying reduction to any
lift recovers the same principal double cover and $J_\chi$, which proves that
all real/Krein conclusions depend only on $[\chi]$.

\subsection{The pressure root}

\begin{lemma}[Irreducible integer matrix at spectral radius one]
\label{lem:app-rho-one-cycle}
Let $A$ be the adjacency matrix of a finite strongly connected directed
multigraph.  Then $\rho(A)=1$ if and only if the graph is a single directed
cycle.  Consequently a strongly connected noncycle graph has $\rho(A)>1$.
\end{lemma}

\begin{proof}
A directed cycle has a permutation adjacency matrix, hence spectral radius one.
Conversely, suppose $\rho(A)=1$.  Let $v>0$ be a Perron vector and choose an
index $i$ with $v_i=\min_jv_j$.  Then
\[
 v_i=(Av)_i=\sum_jA_{ij}v_j
 \ge\left(\sum_jA_{ij}\right)v_i.
\]
The row sum is a positive integer because the graph is strongly connected, so
it must equal one, and its unique successor also has the minimum value $v_i$.
Repeating the argument along successors produces a directed cycle all of whose
vertices have outdegree one.  From $i$ there is only this one forward orbit, but
strong connectivity requires $i$ to reach every vertex.  Hence every vertex
lies on that cycle, and the graph is a single directed cycle.
\end{proof}

For the unweighted depth transfer
\[
  B(s)_{xy}=\sum_{e:x\to y}e^{-s\ell_e},
\]
$B(0)=A$, so \Cref{lem:app-rho-one-cycle} and \textup{(P1)} give
$\rho(B(0))>1$.  If $s_2>s_1$, then every supported edge weight is strictly smaller at
$s_2$.  If $v>0$ is a Perron right vector for $B(s_1)$, every row of the
strongly connected support contains a supported edge, and therefore
\[
  B(s_2)v<B(s_1)v=\rho(B(s_1))v
\]
componentwise.  The strict Collatz--Wielandt inequality gives
$\rho(B(s_2))<\rho(B(s_1))$.  The radius is continuous in $s$ and tends to zero
because every entry does.  The intermediate value theorem therefore gives a
unique positive $\beta_{\rm top}$ with $\rho(B(\beta_{\rm top}))=1$.
The identical argument applies to
$K(s)_{xy}=\sum q_e e^{-s\ell_e}$ with $q_e>0$ whenever $\rho(K(0))>1$.

\subsection{Characterization of the modular magnitude}

On the signed recurrent path cover, let
\[
  Ne_p=\Lambda(p)e_p,
  \qquad \Lambda(pe)=\Lambda(p)+\ell_e.
\]
Then
\[
 e^{\beta N}\widetilde S_e e^{-\beta N}
 =e^{\beta\ell_e}\widetilde S_e.
\]
Let $D_0$ be self-adjoint, diagonal in the signed path basis, deck odd, and
invertible.  Deck oddness gives
$D_0e_{(p,\eta)}=\eta h(p)e_{(p,\eta)}$ with real $h(p)\ne0$.  The vertex
function $a(p)=\sgn h(p)$ is a flat sheet gauge.  Replacing the sheet-sign
representative by $J'_\chi=aJ_\chi$ leaves the principal cover and its holonomy
class unchanged and gives $D_0=J'_\chi|D_0|$.  Rename $J'_\chi$ as $J_\chi$
and put $g=|D_0|>0$.  If
$\Ad(g)(\widetilde S_e)=e^{\beta\ell_e}\widetilde S_e$, then
$m:=ge^{-\beta N}$ commutes with every recurrent lifted shift.  Since the
nontrivial signed cover is connected, a diagonal operator commuting with all
edge shifts is constant on all vertices.  Thus $m=c\one$ and
\[
  D_0=cJ_\chi e^{\beta N}.
\]
Normalizing the magnitude at one base vertex gives $c=1$.  Conversely this
operator has the required deck parity and edge scaling.  If $B$ is bounded and
$J_\chi B^*J_\chi=B$, then
$D=D_0+B$ satisfies $J_\chi D^*J_\chi=D$, so $B$ is precisely a bounded
Krein-self-adjoint differential correction.  These arguments establish all
parts of \Cref{thm:signed-clock}.

\section{Proof of the projective revision algebra}
\label{app:proof-projective-revision}
% =============================================================================

Let $K_{\rev}$ be the elementary abelian two-group of realized reversible
labels acting faithfully on the selected matrix factor $\A_{\rm act}$.  By
\Cref{lem:app-ucp-unit}, every such label acts by a $*$-automorphism.  Every
automorphism of a full complex matrix algebra is inner, so choose unitaries
$R_\alpha$ with $R_0=\one$ and
$\Phi_\alpha=\Ad(R_\alpha)$.

For $\alpha,\beta\in K_{\rev}$, the unitaries
$R_\alpha R_\beta$ and $R_{\alpha+\beta}$ implement the same automorphism.
Their ratio is central, hence scalar:
\[
 R_\alpha R_\beta=\sigma(\alpha,\beta)R_{\alpha+\beta}.
\]
Associativity of the unitary product gives
\[
 \sigma(\alpha,\beta)\sigma(\alpha+\beta,\gamma)
 =\sigma(\beta,\gamma)\sigma(\alpha,\beta+\gamma),
\]
so $\sigma$ is a normalized two-cocycle.  Rephasing $R_\alpha$ changes it by a
coboundary.

The commutator ratio
\[
 b(\alpha,\beta)
 :=\sigma(\alpha,\beta)\sigma(\beta,\alpha)^{-1}
\]
is a bicharacter because $K_{\rev}$ is abelian.  Since every element has order
two,
$b(\alpha,\beta)^2=b(2\alpha,\beta)=1$, and
$b(\alpha,\alpha)=1$.  Hence
$b(\alpha,\beta)=(-1)^{\omega_{\spat}(\alpha,\beta)}$ for a unique alternating
bilinear form $\omega_{\spat}$ over $\F_2$.

The signed sheet action gives
\[
 J_\chi R_\alpha
 =(-1)^{\langle[\chi],\alpha\rangle}R_\alpha J_\chi.
\]
Defining $R_{\alpha+\epsilon t}=R_\alpha J_\chi^\epsilon$ and moving
$J_\chi^\epsilon$ past $R_\beta$ yields
\[
 \omega((\alpha,\epsilon),(\beta,\delta))
 =\omega_{\spat}(\alpha,\beta)
 +\epsilon\langle[\chi],\beta\rangle
 +\delta\langle[\chi],\alpha\rangle.
\]
This proves the first three items of \Cref{thm:projective-revision}.

It remains to identify the primitive algebra.  Let $H$ be the full finite
$\F_2$ label space and let $R=\rad\omega$.  A homogeneous basis element $u_r$
of the abstract twisted group algebra $\C_\sigma[H]$ is central exactly when
$r\in R$.  Fixing a character of this centre reduces the algebra to the twisted
group algebra of $H/R$ with its nondegenerate induced form.  Put
$\dim_{\F_2}(H/R)=2m$ and choose a symplectic basis
$p_1,q_1,\ldots,p_m,q_m$.  After scalar rephasing, the corresponding generators
may be chosen with
\[
 U_j^2=V_j^2=\one,
 \qquad U_jV_j=-V_jU_j,
\]
and generators belonging to distinct pairs commuting.  Each pair generates
$M_2(\C)$, and multiplication gives a surjective homomorphism
\[
 M_2(\C)^{\otimes m}\longrightarrow\C_\sigma[H/R].
\]
Both algebras have complex dimension $2^{2m}$, so this map is an isomorphism:
\[
 \C_\sigma[H/R]\cong M_{2^m}(\C)\cong\Cl_{2m}(\C).
\]
The radical, the quotient, and the matrix type are determined by the derived
commutator form, not separately specified.  This completes the proof.

\section{Proof of the minimal external core}
\label{app:proof-external-core}
% =============================================================================

Let $(H,\omega)=(H_{\rm prim},\omega)$ be nondegenerate of dimension $2m$ and
write $H=K\oplus\F_2t$ as in \textup{(P7)}.

\begin{lemma}[Canonical temporal partner]
\label{lem:app-temporal-partner}
There is a unique nonzero $r\in K$ such that
$\rad(\omega|_K)=K^\perp=\F_2r$.  It satisfies $\omega(t,r)=1$.  Moreover
\[
 K_0:=K\cap t^\perp
\]
is nondegenerate of dimension $2m-2$, and
\[
  H=\Span\{t,r\}\perp K_0,
  \qquad K=\F_2r\oplus K_0.
\]
\end{lemma}

\begin{proof}
Because $K$ is a hyperplane, $\dim K^\perp=1$; let $0\ne r\in K^\perp$.  If
$r\notin K$, then $H=K\oplus\F_2r$ and $r$ is orthogonal to both summands,
contradicting nondegeneracy.  Thus $r\in K$ and
$K\cap K^\perp=\F_2r$.  If $\omega(t,r)=0$, then $r$ is orthogonal to $K$ and
to $t$, again a contradiction; over $\F_2$ the nonzero value is one.

For $k\in K$, set $k_0=k+\omega(t,k)r$.  Then $k_0\in K_0$, so
$K=\F_2r\oplus K_0$.  If $x\in K_0$ is orthogonal to $K_0$, then it is also
orthogonal to $r$ because $r\in\rad(\omega|_K)$ and to $t$ by definition.
Hence $x$ is orthogonal to all of $H$ and is zero.  This proves nondegeneracy and
the orthogonal decomposition.
\end{proof}

By \textup{(P7)}, $K_0\ne0$.  Choose $0\ne b\in K_0$.  Nondegeneracy gives
$c\in K_0$ with $\omega(b,c)=1$.  The planes
$P_t=\Span\{t,r\}$ and $P_s=\Span\{b,c\}$ are hyperbolic and orthogonal, so
\[
 W_{\ext}=P_t\perp P_s
\]
is nondegenerate of dimension four.  Since $t\notin K$ and $r,b,c\in K$,
\[
 W_{\ext}\cap K=\Span\{r,b,c\}
\]
has dimension three.  The orthogonal complement
$H_{\intr}=W_{\ext}^\perp$ is nondegenerate of dimension $2m-4$ and lies in
$K_0$: if $x=k+\epsilon t\in W_{\ext}^\perp$, then
$0=\omega(x,r)=\epsilon$, and $0=\omega(x,t)$ then puts $x\in K_0$.
Thus
\[
 H=W_{\ext}\perp H_{\intr},
 \qquad
 K=(W_{\ext}\cap K)\oplus H_{\intr}.
\]

The restrictions of $\omega$ to the two summands are nondegenerate.  The proof
of \Cref{thm:projective-revision} therefore gives
\[
 \A_{\ext}\cong M_4(\C),
 \qquad
 \A_{\intr}\cong M_{2^{m-2}}(\C).
\]
Orthogonality makes their homogeneous generators commute.  Every label has a
unique decomposition $w+v$, so multiplication gives a surjective map
$\A_{\ext}\otimes\A_{\intr}\to\A_{\rm act}$.  The dimensions agree:
\[
 16\cdot 2^{2m-4}=2^{2m}=\dim_\C\A_{\rm act},
\]
hence the map is an isomorphism.

Normalize $R_t=J_\chi$, $R_r$, $R_b$, and $R_c$ as Hermitian involutions.
The only nontrivial commutators among the two hyperbolic pairs are
$R_tR_r=-R_rR_t$ and $R_bR_c=-R_cR_b$.  Then
\[
 \gamma^0:=R_tR_r
\]
is anti-Hermitian and has square $-\one$.  An explicit compatible spatial
Clifford family is
\[
 \gamma^1=R_tR_b,
 \qquad
 \gamma^2=R_tR_c,
 \qquad
 \gamma^3=iR_tR_bR_c.
\]
Each is Hermitian, each squares to $\one$, they anticommute pairwise, and each
anticommutes with $\gamma^0$; these identities follow directly by moving the
four generators past one another with the two displayed signs.  Thus the
external factor is a $(1,3)$ Clifford module.

Finally, $\dim K=2m-1$ is odd.  If $m=1$, then $\dim K=1$ and
$\bigwedge^2K=0$, so there is no nonzero spatial two-reset sector.  The next
case is $m=2$, where $\dim K=3$ and the above construction gives the full
external factor.  Hence three is the minimal spatially nondegenerate rank,
while $H_{\intr}$ may remain nonzero.  This proves
\Cref{thm:external-core}.

\section{Proof of the canonical external direction record}
\label{app:proof-octahedral-record}

\begin{proof}[Detailed proof of \Cref{thm:octahedral-record}]
Because each $\gamma^i$ is a Hermitian involution,
$P_{i,\pm}=(\one\pm\gamma^i)/2$ are orthogonal projections and
$P_{i,+}+P_{i,-}=\one$.  Hence
\[
 \sum_{i,\pm}K_{i,\pm}^*K_{i,\pm}
 =\frac13\sum_i(P_{i,+}+P_{i,-})=\one,
\]
so the six branch maps form a trace-preserving resolved instrument.

Let $p_{i,\pm}(\rho)=\Tr(\rho E_{i,\pm})$.  Since
$\theta_{i,\pm}\theta_{i,\pm}^{\mathsf T}=e_ie_i^{\mathsf T}$,
\[
 \begin{aligned}
 \sum_{i,\pm}p_{i,\pm}(\rho)
 \theta_{i,\pm}\theta_{i,\pm}^{\mathsf T}
 &=\sum_i\Tr\!\left(\rho\frac{P_{i,+}+P_{i,-}}3\right)e_ie_i^{\mathsf T}\\
 &=\frac13\sum_i e_ie_i^{\mathsf T}=\frac13I_3.
 \end{aligned}
\]
Thus the second moment is state independent and positive definite.

Write $\gamma(\theta_{i,\pm})=\pm\gamma^i$.  The two signs cancel in
$\gamma(\theta)(\theta\cdot\xi)$, and therefore
\[
 \sum_{i,\pm}p_{i,\pm}(\rho)
 \gamma(\theta_{i,\pm})(\theta_{i,\pm}\cdot\xi)
 =\frac13\sum_i\gamma^i\xi_i.
\]
With frame normalization three, the symbol is the one stated in the theorem.
Using $(\gamma^0)^2=-\one$, $(\gamma^i)^2=\one$, and all mixed
anticommutators equal to zero,
\[
 \left[i\left(\gamma^0\xi_0+\sum_i\gamma^i\xi_i\right)\right]^2
 =\left(\xi_0^2-|\vec\xi|^2\right)\one.
\]
\end{proof}

\section{Proof of the flat continuum theorem}
\label{app:proof-flat-limit}
% =============================================================================

Put $A^i=\gamma^0\gamma^i$.  The Clifford relations give
$(A^i)^*=A^i$ and
\[
 \{A^i,A^j\}=2\delta^{ij}\one.
\]
For a direction $\theta\in\Sphere^2$, write
$A(\theta)=\theta_iA^i$.

\subsection{Symbol and frame equivalence}

Let $v_i=\kappa M^{ij}\xi_j$.  Since the tensor $v_iv_j$ is symmetric,
\[
 \begin{aligned}
 \sigma_M(\xi)^2
 &=-\left(\gamma^0\xi_0+\gamma^iv_i\right)^2\\
 &=-\left((\gamma^0)^2\xi_0^2
 +\tfrac12\{\gamma^i,\gamma^j\}v_iv_j\right)\\
 &=\left(\xi_0^2-|v|^2\right)\one
 =\left(\xi_0^2-\kappa^2\xi^{\mathsf T}M^2\xi\right)\one.

\subsection{Discrete fundamental theorem}

$C_{ade}+\gamma_{d;ae}-\gamma_{e;ad}=0$, which the displayed formula satisfies.

 \Delta_{d;ae}=\Delta_{e;ad}=-\Delta_{e;da}
 =-\Delta_{a;de}=\Delta_{a;ed}=\Delta_{d;ea}
